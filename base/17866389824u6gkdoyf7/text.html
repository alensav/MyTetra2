🔧 НАСТРОЙКА NGINX (ДЛЯ HTTPS И MINIO)

bash

# Узнайте IP-адреса контейнеров

FRONTEND_IP=$(docker inspect -f '{{range .NetworkSettings.Networks}}{{.IPAddress}}{{end}}' alexandrie-frontend)

BACKEND_IP=$(docker inspect -f '{{range .NetworkSettings.Networks}}{{.IPAddress}}{{end}}' alexandrie-backend)

MINIO_IP=$(docker inspect -f '{{range .NetworkSettings.Networks}}{{.IPAddress}}{{end}}' alexandrie-rustfs)


# Создайте конфиг nginx

cat > /etc/nginx/sites-available/alexandrie << EOF

server {

listen 8200 ssl;

server_name 78.17.187.53.nip.io;


ssl_certificate /etc/letsencrypt/live/78.17.187.53.nip.io/fullchain.pem;

ssl_certificate_key /etc/letsencrypt/live/78.17.187.53.nip.io/privkey.pem;


location / {

proxy_pass http://${FRONTEND_IP}:3000;

proxy_set_header Host \$host;

proxy_set_header X-Real-IP \$remote_addr;

proxy_set_header X-Forwarded-For \$proxy_add_x_forwarded_for;

proxy_set_header X-Forwarded-Proto \$scheme;

}


location /api/ {

proxy_pass http://${BACKEND_IP}:8201/api/;

proxy_set_header Host \$host;

proxy_set_header X-Real-IP \$remote_addr;

proxy_set_header X-Forwarded-For \$proxy_add_x_forwarded_for;

proxy_set_header X-Forwarded-Proto \$scheme;

}


# Прокси для MinIO (чтобы файлы открывались по HTTPS)

location /alexandrie/ {

proxy_pass http://${MINIO_IP}:9000/alexandrie/;

proxy_set_header Host \$host;

proxy_set_header X-Real-IP \$remote_addr;

proxy_set_header X-Forwarded-For \$proxy_add_x_forwarded_for;

proxy_set_header X-Forwarded-Proto \$scheme;

}

}

EOF


# Включите конфиг и перезапустите nginx

ln -sf /etc/nginx/sites-available/alexandrie /etc/nginx/sites-enabled/

nginx -t

systemctl restart nginx
