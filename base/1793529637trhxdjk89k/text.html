Шаг 2. Скрипт /root/scripts/monitor_proekts.sh

#!/bin/bash
# Мониторинг магазина proekts
set -uo pipefail

BOT_TOKEN="7123456789:AAHdqTcvCH1vGWJxfSeofSAs0K5PALDsaw"
CHAT_ID="123456789"
LOG="/var/log/monitor_proekts.log"

log() { echo "$(date '+%Y-%m-%d %H:%M:%S') $*" >> "$LOG"; }
tg_send() {
    curl -sS -X POST "https://api.telegram.org/bot${BOT_TOKEN}/sendMessage" \
        -d "chat_id=${CHAT_ID}" -d "parse_mode=HTML" -d "text=${1}" > /dev/null 2>&1
}

check_site() {
    local code
    code=$(curl -sS -o /dev/null -w "%{http_code}" --max-time 15 \
        "https://proekts.alensav.ru/" 2>/dev/null || echo "000")
    if [[ "$code" != "200" ]]; then
        log "ALERT: сайт вернул HTTP $code"
        tg_send "🚨 <b>Сайт недоступен</b>"
        return 1
    fi
}

check_container() {
    local status
    status=$(cd /opt/yookassa-to-mynalog && sudo docker compose ps --format '{{.Status}}' 2>/dev/null | head -1)
    if [[ "$status" != *"Up"* ]]; then
        log "ALERT: контейнер $status"
        tg_send "🚨 <b>Контейнер yookassa-to-mynalog не работает</b>"
        return 1
    fi
}

check_sshfs() {
    if ! mount | grep -q "orangepi.*fuse.sshfs"; then
        log "ALERT: SSHFS не смонтирован"
        tg_send "🚨 <b>SSHFS не смонтирован</b>"
        return 1
    fi
    if ! sudo -u www-data test -r /mnt/external/Proekts/TEST.txt 2>/dev/null; then
        log "ALERT: SSHFS смонтирован, но файлы не читаются"
        tg_send "🚨 <b>SSHFS не читает файлы</b>"
    fi
}

check_disk() {
    local usage
    usage=$(df / | awk 'NR==2 {print $5}' | tr -d '%')
    if [[ "$usage" -gt 85 ]]; then
        log "ALERT: диск занят $usage%"
        tg_send "🚨 <b>Мало места на диске</b>%0AЗанято: ${usage}%25"
    fi
}

check_load() {
    local load
    load=$(awk '{print $1}' /proc/loadavg)
    local load_int=${load%.*}
    if [[ "$load_int" -gt 5 ]]; then
        log "WARN: load average $load"
        tg_send "⚠️ <b>Высокий load average</b>%0ALoad: ${load}"
    fi
}

log "=== START ==="
check_site
check_container
check_sshfs
check_disk
check_load
log "=== DONE ==="

Ротация: ежедневные — 30 дней; еженедельные — 12 недель.

Скрипт /root/scripts/backup_proekts.sh

#!/bin/bash
# Автобэкап магазина PROEKTS
set -uo pipefail

DATE=$(date +%F)
WEEK=$(date +%Y-W%V)
BACKUP_DIR="/root/backups/proekts"
LOG="/var/log/backup_proekts.log"

mkdir -p "$BACKUP_DIR"
cd "$BACKUP_DIR"

log() { echo "$(date '+%Y-%m-%d %H:%M:%S') $*" >> "$LOG"; }

log "=== START ==="

mysqldump --single-transaction --quick --routines --triggers \
    --default-character-set=utf8mb4 proekts | gzip > "mysql-proekts-${DATE}.sql.gz"

docker compose -f /opt/yookassa-to-mynalog/docker-compose.yml \
    exec -T -u app yookassa-to-mynalog tar czf - -C /app data \
    > "container-data-${DATE}.tgz" 2>/dev/null

cp /opt/yookassa-to-mynalog/.env "container-env-${DATE}.env"
chmod 600 "container-env-${DATE}.env"

if [[ "$(date +%u)" == "7" ]]; then
    tar czf "files-${WEEK}.tgz" -C /var/www proekts.alensav.ru \
        --exclude='*.bak*' --exclude='vendor/*/tests' 2>/dev/null
    tar czf "nginx-${WEEK}.tgz" -C /etc nginx/sites-available nginx/sites-enabled nginx/nginx.conf 2>/dev/null
    tar czf "systemd-${WEEK}.tgz" \
        /etc/systemd/system/mnt-orangepi-proekts.service \
        /etc/systemd/system/frps.service 2>/dev/null
fi

find "$BACKUP_DIR" -maxdepth 1 -name "mysql-proekts-*.sql.gz" -mtime +30 -delete
find "$BACKUP_DIR" -maxdepth 1 -name "container-data-*.tgz" -mtime +30 -delete
find "$BACKUP_DIR" -maxdepth 1 -name "container-env-*.env" -mtime +30 -delete
find "$BACKUP_DIR" -maxdepth 1 -name "files-*.tgz" -mtime +90 -delete
find "$BACKUP_DIR" -maxdepth 1 -name "nginx-*.tgz" -mtime +90 -delete
find "$BACKUP_DIR" -maxdepth 1 -name "systemd-*.tgz" -mtime +90 -delete

log "=== DONE ==="

Cron — автозапуск

# /etc/cron.d/backup-proekts
0 3 * * * root /root/scripts/backup_proekts.sh

18.3. Восстановление из бэкапа — полный регламент

Сценарий 1. Восстановить сайт полностью

  1. Восстановить ОС + LEMP (nginx, php8.3-fpm, mariadb, docker).
  2. Развернуть бэкап-файлы из внешнего хранилища.
  3. Восстановить MySQL из дампа.
  4. Восстановить файлы сайта из tar.gz.
  5. Восстановить nginx-конфиги.
  6. Восстановить SSL через certbot.
  7. Восстановить systemd-сервисы (frps, mnt-orangepi-proekts).
  8. Восстановить контейнер (SQLite + .env).
  9. Восстановить SSH-ключ для OrangePi.
  10. Проверить: сайт, контейнер, SSHFS, рассылку.

Сценарий 2. Восстановить только MySQL

sudo mysqldump --single-transaction proekts | gzip > /tmp/proekts-before-restore.sql.gz
gunzip < /root/backups/proekts/mysql-proekts-2026-10-05.sql.gz | sudo mysql proekts

Сценарий 3. Восстановить только SQLite контейнера

cd /opt/yookassa-to-mynalog
sudo docker compose down
sudo tar xzf /root/backups/proekts/container-data-2026-10-06.tgz -C /tmp/container-restore
sudo docker compose up -d
sudo docker compose cp /tmp/container-restore/data/. yookassa-to-mynalog:/app/data/
sudo docker compose exec -u root yookassa-to-mynalog chown -R app:app /app/data

Сценарий 4. Восстановить один файл

sudo tar xzf /root/backups/proekts/files-2026-W40.tgz -C /var/www proekts.alensav.ru/index.php
sudo chown www-data:www-data /var/www/proekts.alensav.ru/index.php
