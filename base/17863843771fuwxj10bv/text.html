📋 ИТОГОВАЯ РАБОЧАЯ ИНСТРУКЦИЯ ПО ЗАПУСКУ ALEXANDRIE

✅ 1. Подготовка системы

bash

# Очистка памяти (если нужно)

sudo docker system prune -a -f --volumes


# Создание сети

sudo docker network create alexandrie-net 2>/dev/null

✅ 2. Запуск MySQL

bash

sudo docker run -d \

--name alexandrie-mysql \

--network alexandrie-net \

--restart=always \

-p 3307:3306 \

-e MYSQL_DATABASE=alexandrie \

-e MYSQL_USER=alexandrie \

-e MYSQL_PASSWORD=password \

-e MYSQL_ROOT_PASSWORD=rootpassword \

mysql:8.0.36

✅ 3. Запуск бэкенда

bash

sudo docker run -d \

--name alexandrie-backend \

--network alexandrie-net \

--restart=always \

-p 8201:8201 \

-e BACKEND_PORT=8201 \

-e COOKIE_DOMAIN=localhost \

-e FRONTEND_URL=http://localhost:8200 \

-e DATABASE_HOST=alexandrie-mysql \

-e DATABASE_PORT=3306 \

-e DATABASE_USER=alexandrie \

-e DATABASE_PASSWORD=password \

-e DATABASE_NAME=alexandrie \

-e MINIO_ENDPOINT=localhost:9000 \

-e JWT_SECRET=your-secure-jwt-secret-key-change-this-in-production \

ghcr.io/smaug6739/alexandrie-backend:latest

✅ 4. Запуск фронтенда (без проброса порта)

bash

sudo docker run -d \

--name alexandrie-frontend \

--network alexandrie-net \

--restart=always \

ghcr.io/smaug6739/alexandrie-frontend:latest

✅ 5. Настройка nginx (прокси)

Создайте файл ~/nginx.conf:

bash

cat > ~/nginx.conf << 'EOF'

events {

worker_connections 1024;

}


http {

server {

listen 8200;


location / {

proxy_pass http://alexandrie-frontend:3000;

proxy_set_header Host $host;

proxy_set_header X-Real-IP $remote_addr;

proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;

proxy_set_header X-Forwarded-Proto $scheme;

}


location /api/ {

proxy_pass http://alexandrie-backend:8201/api/;

proxy_set_header Host $host;

proxy_set_header X-Real-IP $remote_addr;

proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;

proxy_set_header X-Forwarded-Proto $scheme;

}

}

}

EOF

✅ 6. Запуск nginx

bash

sudo docker run -d \

--name alexandrie-nginx \

--network alexandrie-net \

--restart=always \

-p 8200:8200 \

-v ~/nginx.conf:/etc/nginx/nginx.conf:ro \

nginx:alpine

alexandrie-frontend

✅ 8. Создание пользователя

bash

curl -X POST http://localhost:8201/api/users \

-H "Content-Type: application/json" \

-d '{

"username": "alensav",

"email": "sav27951@yandex.ru",

"password": "admin123"

}'

✅ 9. Проверка авторизации

bash

curl -X POST http://localhost:8201/api/auth \

-H "Content-Type: application/json" \

-d '{

"username": "alensav",

"password": "admin123"

}'

Ожидаемый ответ — данные пользователя.

✅ 10. Вход в систему

Откройте в браузере: http://localhost:8200

  • Username: alensav
  • Password: admin123

✅ 11. Остановка всех контейнеров

bash

sudo docker stop alexandrie-mysql alexandrie-backend alexandrie-frontend alexandrie-nginx

✅ 12. Удаление всех контейнеров

bash

sudo docker rm alexandrie-mysql alexandrie-backend alexandrie-frontend alexandrie-nginx

✅ 13. Полная очистка

bash

sudo docker system prune -a -f --volumes

Примечания

  • nginx обязателен, т.к. образ фронтенда игнорирует переменные окружения.
  • Все контейнеры должны быть в одной сети (alexandrie-net).
  • Для входа использовать только localhost:8200 (не IP:8200).
  • Если что-то сломалось — повторить инструкцию с шага 1.
